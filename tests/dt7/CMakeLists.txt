cmake_minimum_required(VERSION 3.22)
project(dt7_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(DT7_DIR "${ROOT}/User_File/3_Modules/dt7")
add_executable(dt7_tests dt7_tests.cpp "${DT7_DIR}/dt7_protocol.cpp")
target_include_directories(dt7_tests PRIVATE "${DT7_DIR}")
target_compile_options(dt7_tests PRIVATE -Wall -Wextra -Werror)
enable_testing()
foreach(scenario channels switches splits invalid resync timeout partial_timeout wrap_reset)
    add_test(NAME dt7_${scenario} COMMAND dt7_tests ${scenario})
endforeach()
